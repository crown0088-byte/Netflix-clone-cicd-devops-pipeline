pipeline {
    agent any
    
    environment {
        SCANNER_HOME = tool 'sonar-scanner'
    }
    
    stages {
        stage('clean workspace') {
            steps {
                cleanWs()
            }
        }
        
        stage('Checkout from Git') {
            steps {
                git branch: 'main', url: 'https://github.com/crown0088-byte/netflix-clone.git'
            }
        }
        
        stage('Sonarqube Analysis') {
            steps {
                withSonarQubeEnv('sonar-server') {
                    sh '''
                    $SCANNER_HOME/bin/sonar-scanner \
                    -Dsonar.projectKey=Netflix \
                    -Dsonar.projectName=Netflix
                    '''
                }
            }
        }
        
        stage('quality gate') {
            steps {
                script {
                    waitForQualityGate abortPipeline: true
                }
            }
        }
        
        stage('Install Dependencies') {
            steps {
                sh 'npm install'
            }
        }
        
        stage('TRIVY FS SCAN') {
            steps {
                sh 'trivy fs . > trivy-fs-report.txt'
            }
        }
        
        stage('Docker Build & Push') {
            steps {
                script {
                    sh "sed -i 's/FROM node:16.*/FROM node:18-alpine/g' Dockerfile"
                    sh 'docker build -t byte1575/netflix:latest .'
                }
            }
        }
        
        stage('TRIVY') {
            steps {
                sh 'trivy image byte1575/netflix:latest > trivy-image-report.txt'
            }
        }
        
        stage('Push to Registry') {
            steps {
                withCredentials([string(credentialsId: 'docker', variable: 'docker_password')]) {
                    sh 'docker login -u byte1575 -p ${docker_password}'
                    sh 'docker push byte1575/netflix:latest'
                }
            }
        }
        
        stage('Deploy to container') {
            steps {
                sh 'docker rm -f netflix-app || true'
                sh 'docker run -d --name netflix-app -p 8081:80 byte1575/netflix:latest'
            }
        }
    }
}
